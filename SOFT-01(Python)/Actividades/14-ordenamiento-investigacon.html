Investigación: algoritmos de ordenamiento

Propósito

Ordenar datos es una de las tareas más comunes en programación: ordenar notas, precios, nombres o fechas. Existen muchas formas de hacerlo y no todas tardan lo mismo. En esta actividad usted investiga cinco algoritmos de ordenamiento, los explica con sus palabras y luego comprueba con programas en Python si lo que encontró es cierto.

  • Bubble sort o burbuja
  • Selection sort o selección
  • Insertion sort o inserción
  • Merge sort o mezcla
  • Quicksort o rápido

01_burbuja.py 
# ============================================================
# Archivo: 01_burbuja.py
# Curso: SOFT-01 Principios de Programación 1
# Descripción: Ordenamiento por burbuja (Bubble sort): compara elementos vecinos y los
#              intercambia; en cada pasada el mayor "sube" al final.
# Ejecución: python 01_burbuja.py
# ============================================================
import random
import time

# Tamaños de lista que se usan para medir el tiempo
TAMANOS = (100, 1000, 3000)
LISTA_EJEMPLO = [5, 2, 8, 1, 9, 3]


def burbuja(lista, mostrar):
    """Ordena la lista con burbuja. Devuelve (lista, comparaciones, movimientos)."""
    comparaciones = 0
    movimientos = 0
    n = len(lista)
    hubo_intercambio = True
    pasada = 0
    # Se repite mientras en la pasada anterior haya habido algún intercambio
    while hubo_intercambio:
        hubo_intercambio = False
        pasada = pasada + 1
        for j in range(n - pasada):
            comparaciones = comparaciones + 1
            if lista[j] > lista[j + 1]:
                lista[j], lista[j + 1] = lista[j + 1], lista[j]
                movimientos = movimientos + 1
                hubo_intercambio = True
        if mostrar:
            print(f"  Pasada {pasada}: {lista}")
    return lista, comparaciones, movimientos


def medir(lista):
    """Ordena una copia de la lista y devuelve (segundos, comparaciones, movimientos)."""
    copia = lista[:]                      # se ordena una copia para no alterar la original
    inicio = time.perf_counter()
    resultado, comparaciones, movimientos = burbuja(copia, False)
    fin = time.perf_counter()
    if resultado != sorted(lista):        # comprobación: el resultado debe estar ordenado
        print("¡Error! La lista no quedó ordenada.")
    return fin - inicio, comparaciones, movimientos


def main():
    print("=" * 60)
    print("ORDENAMIENTO POR BURBUJA (BUBBLE SORT)")
    print("=" * 60)

    # 1. Ejemplo paso a paso con una lista pequeña
    print("\nLista inicial:", LISTA_EJEMPLO)
    resultado, comparaciones, movimientos = burbuja(LISTA_EJEMPLO[:], True)
    print("Lista ordenada:", resultado)
    print("Comparaciones:", comparaciones, "| Movimientos:", movimientos)

    # 2. Medición de tiempo con listas aleatorias de distintos tamaños
    random.seed(2026)                     # misma semilla: todos obtienen las mismas listas
    print("\nTiempo con listas aleatorias:")
    print(f"{'Tamaño':>8} {'Segundos':>12} {'Comparaciones':>15} {'Movimientos':>13}")
    for tamano in TAMANOS:
        lista = [random.randint(1, 10000) for _ in range(tamano)]
        segundos, comparaciones, movimientos = medir(lista)
        print(f"{tamano:>8} {segundos:>12.5f} {comparaciones:>15} {movimientos:>13}")

    # 3. Mejor y peor caso con 1000 elementos
    print("\nCon 1000 elementos según el orden inicial:")
    casos = (("Ya ordenada", list(range(1000))),
             ("Orden inverso", list(range(1000, 0, -1))),
             ("Aleatoria", [random.randint(1, 10000) for _ in range(1000)]))
    for nombre, lista in casos:
        segundos, comparaciones, movimientos = medir(lista)
        print(f"{nombre:>14}: {segundos:.5f} s | {comparaciones} comparaciones | {movimientos} movimientos")


main()

import random
import time

# Tamaños de lista que se usan para medir el tiempo
TAMANOS = (100, 1000, 3000)
LISTA_EJEMPLO = [5, 2, 8, 1, 9, 3]


def seleccion(lista, mostrar):
    """Ordena la lista con selección. Devuelve (lista, comparaciones, movimientos)."""
    comparaciones = 0
    movimientos = 0
    n = len(lista)
    for i in range(n - 1):
        # Buscar la posición del menor entre i y el final
        pos_menor = i
        for j in range(i + 1, n):
            comparaciones = comparaciones + 1
            if lista[j] < lista[pos_menor]:
                pos_menor = j
        # Llevar el menor a la posición i
        if pos_menor != i:
            lista[i], lista[pos_menor] = lista[pos_menor], lista[i]
            movimientos = movimientos + 1
        if mostrar:
            print(f"  Paso {i + 1}: menor = {lista[i]} -> {lista}")
    return lista, comparaciones, movimientos


def medir(lista):
    """Ordena una copia de la lista y devuelve (segundos, comparaciones, movimientos)."""

# Descripción: Ordenamiento por inserción (Insertion sort): toma cada elemento y lo
#              inserta en su lugar dentro de la parte ya ordenada.
# Ejecución: python 03_insercion.py
# ============================================================
import random
import time

# Tamaños de lista que se usan para medir el tiempo
TAMANOS = (100, 1000, 3000)
LISTA_EJEMPLO = [5, 2, 8, 1, 9, 3]


def insercion(lista, mostrar):
    """Ordena la lista con inserción. Devuelve (lista, comparaciones, movimientos)."""
    comparaciones = 0
    movimientos = 0
    for i in range(1, len(lista)):
        actual = lista[i]
        j = i - 1
        # Correr hacia la derecha los elementos mayores que 'actual'
        seguir = True
        while j >= 0 and seguir:
            comparaciones = comparaciones + 1
            if lista[j] > actual:
                lista[j + 1] = lista[j]
                movimientos = movimientos + 1
                j = j - 1
            else:
                seguir = False
        lista[j + 1] = actual
        if mostrar:
            print(f"  Insertar {actual}: {lista}")
    return lista, comparaciones, movimientos


def medir(lista):
    """Ordena una copia de la lista y devuelve (segundos, comparaciones, movimientos)."""
    copia = lista[:]                      # se ordena una copia para no alterar la original
    inicio = time.perf_counter()
    resultado, comparaciones, movimientos = insercion(copia, False)
    fin = time.perf_counter()
    if resultado != sorted(lista):        # comprobación: el resultado debe estar ordenado
        print("¡Error! La lista no quedó ordenada.")
    return fin - inicio, comparaciones, movimientos

# Archivo: 04_mezcla.py
# Curso: SOFT-01 Principios de Programación 1
# Descripción: Ordenamiento por mezcla (Merge sort): divide la lista en mitades, ordena
#              cada mitad y luego las mezcla en orden.
# Ejecución: python 04_mezcla.py
# ============================================================
import random
import time

# Tamaños de lista que se usan para medir el tiempo
TAMANOS = (100, 1000, 3000)
LISTA_EJEMPLO = [5, 2, 8, 1, 9, 3]


def mezclar(izquierda, derecha, contador):
    """Une dos listas ordenadas en una sola lista ordenada."""
    resultado = []
    i = 0
    j = 0
    while i < len(izquierda) and j < len(derecha):
        contador[0] = contador[0] + 1                 # una comparación
        if izquierda[i] <= derecha[j]:
            resultado.append(izquierda[i])
            i = i + 1
        else:
            resultado.append(derecha[j])
            j = j + 1
        contador[1] = contador[1] + 1                 # un movimiento
    # Agregar lo que sobre de cualquiera de las dos mitades
    while i < len(izquierda):
        resultado.append(izquierda[i])
        i = i + 1
        contador[1] = contador[1] + 1
    while j < len(derecha):
        resultado.append(derecha[j])
        j = j + 1
        contador[1] = contador[1] + 1
    return resultado


def dividir_y_mezclar(lista, contador, mostrar, nivel):
    """Función recursiva: se llama a sí misma con cada mitad."""
    if len(lista) <= 1:
        resultado = lista
    else:
        mitad = len(lista) // 2
        izquierda = dividir_y_mezclar(lista[:mitad], contador, mostrar, nivel + 1)
        derecha = dividir_y_mezclar(lista[mitad:], contador, mostrar, nivel + 1)
        resultado = mezclar(izquierda, derecha, contador)
        if mostrar:
            print("  " + "  " * nivel + f"Mezclar {izquierda} + {derecha} -> {resultado}")
    return resultado


def mezcla(lista, mostrar):
    """Ordena la lista con mezcla. Devuelve (lista, comparaciones, movimientos)."""
    contador = [0, 0]                                 # [comparaciones, movimientos]
    resultado = dividir_y_mezclar(lista, contador, mostrar, 0)
    return resultado, contador[0], contador[1]


def medir(lista):
    """Ordena una copia de la lista y devuelve (segundos, comparaciones, movimientos)."""
    copia = lista[:]                      # se ordena una copia para no alterar la original
    inicio = time.perf_counter()
    resultado, comparaciones, movimientos = mezcla(copia, False)
    fin = time.perf_counter()
    if resultado != sorted(lista):        # comprobación: el resultado debe estar ordenado
        print("¡Error! La lista no quedó ordenada.")
    return fin - inicio, comparaciones, movimientos

# Archivo: 05_rapido.py
# Curso: SOFT-01 Principios de Programación 1
# Descripción: Ordenamiento rápido (Quicksort): elige un pivote, deja a la izquierda los
#              menores y a la derecha los mayores, y repite en cada parte.
# Ejecución: python 05_rapido.py
# ============================================================
import random
import time

# Tamaños de lista que se usan para medir el tiempo
TAMANOS = (100, 1000, 3000)
LISTA_EJEMPLO = [5, 2, 8, 1, 9, 3]


def particionar(lista, inicio, fin, contador):
    """Usa el elemento del medio como pivote; devuelve su posición final."""
    medio = (inicio + fin) // 2
    lista[medio], lista[fin] = lista[fin], lista[medio]   # el pivote pasa al final
    pivote = lista[fin]
    posicion = inicio
    for j in range(inicio, fin):
        contador[0] = contador[0] + 1
        if lista[j] < pivote:
            lista[posicion], lista[j] = lista[j], lista[posicion]
            contador[1] = contador[1] + 1
            posicion = posicion + 1
    lista[posicion], lista[fin] = lista[fin], lista[posicion]
    contador[1] = contador[1] + 1
    return posicion


def ordenar_parte(lista, inicio, fin, contador, mostrar):
    """Función recursiva: ordena la parte de la lista entre inicio y fin."""
    if inicio < fin:
        p = particionar(lista, inicio, fin, contador)
        if mostrar:
            print(f"  Pivote {lista[p]}: {lista[inicio:p]} < {lista[p]} < {lista[p + 1:fin + 1]}")
        ordenar_parte(lista, inicio, p - 1, contador, mostrar)
        ordenar_parte(lista, p + 1, fin, contador, mostrar)


def rapido(lista, mostrar):
    """Ordena la lista con quicksort. Devuelve (lista, comparaciones, movimientos)."""
    contador = [0, 0]                                     # [comparaciones, movimientos]
    ordenar_parte(lista, 0, len(lista) - 1, contador, mostrar)
    return lista, contador[0], contador[1]


def medir(lista):
    """Ordena una copia de la lista y devuelve (segundos, comparaciones, movimientos)."""
    copia = lista[:]                      # se ordena una copia para no alterar la original
    inicio = time.perf_counter()
    resultado, comparaciones, movimientos = rapido(copia, False)
    fin = time.perf_counter()
    if resultado != sorted(lista):        # comprobación: el resultado debe estar ordenado
        print("¡Error! La lista no quedó ordenada.")
    return fin - inicio, comparaciones, movimientos


def main():
    print("=" * 60)
    print("ORDENAMIENTO RÁPIDO (QUICKSORT)")
    print("=" * 60)

    # 1. Ejemplo paso a paso con una lista pequeña
    print("\nLista inicial:", LISTA_EJEMPLO)
    resultado, comparaciones, movimientos = rapido(LISTA_EJEMPLO[:], True)
    print("Lista ordenada:", resultado)
    print("Comparaciones:", comparaciones, "| Movimientos:", movimientos)

    print("\nTiempo con listas aleatorias:")
    print(f"{'Tamaño':>8} {'Segundos':>12} {'Comparaciones':>15} {'Movimientos':>13}")
    for tamano in TAMANOS:
        lista = [random.randint(1, 10000) for _ in range(tamano)]
        segundos, comparaciones, movimientos = medir(lista)
        print(f"{tamano:>8} {segundos:>12.5f} {comparaciones:>15} {movimientos:>13}")

        for j in range(i + 1, n):
            comparaciones = comparaciones + 1
            if lista[j] < lista[pos_menor]:
                pos_menor = j
        # Llevar el menor a la posición i
        if pos_menor != i:
            lista[i], lista[pos_menor] = lista[pos_menor], lista[i]
            movimientos = movimientos + 1
        if mostrar:
            print(f"  Paso {i + 1}: menor = {lista[i]} -> {lista}")
    return lista, comparaciones, movimientos

# ---------- Ordenamiento Por Inserción (Insertion Sort) ----------

def insercion(lista, mostrar):
    """Ordena la lista con inserción. Devuelve (lista, comparaciones, movimientos)."""
    comparaciones = 0
    movimientos = 0
    for i in range(1, len(lista)):
        actual = lista[i]
        j = i - 1
        # Correr hacia la derecha los elementos mayores que 'actual'
        seguir = True
        while j >= 0 and seguir:
            comparaciones = comparaciones + 1
            if lista[j] > actual:
                lista[j + 1] = lista[j]
                movimientos = movimientos + 1
                j = j - 1
            else:
                seguir = False
        lista[j + 1] = actual
        if mostrar:
            print(f"  Insertar {actual}: {lista}")
    return lista, comparaciones, movimientos

# ---------- Ordenamiento Por Mezcla (Merge Sort) ----------

def mezclar(izquierda, derecha, contador):
    """Une dos listas ordenadas en una sola lista ordenada."""
    resultado = []
    i = 0
    j = 0
    while i < len(izquierda) and j < len(derecha):
        contador[0] = contador[0] + 1                 # una comparación
        if izquierda[i] <= derecha[j]:
            resultado.append(izquierda[i])
            i = i + 1
        else:
            resultado.append(derecha[j])
            j = j + 1
        contador[1] = contador[1] + 1                 # un movimiento
    # Agregar lo que sobre de cualquiera de las dos mitades
    while i < len(izquierda):
        resultado.append(izquierda[i])
        i = i + 1
        contador[1] = contador[1] + 1
    while j < len(derecha):
        resultado.append(derecha[j])
        j = j + 1
        contador[1] = contador[1] + 1
    return resultado


def dividir_y_mezclar(lista, contador, mostrar, nivel):
    """Función recursiva: se llama a sí misma con cada mitad."""
    if len(lista) <= 1:
        resultado = lista
    else:
        mitad = len(lista) // 2
        izquierda = dividir_y_mezclar(lista[:mitad], contador, mostrar, nivel + 1)
        derecha = dividir_y_mezclar(lista[mitad:], contador, mostrar, nivel + 1)
        resultado = mezclar(izquierda, derecha, contador)
        if mostrar:
            print("  " + "  " * nivel + f"Mezclar {izquierda} + {derecha} -> {resultado}")
    return resultado


def mezcla(lista, mostrar):
    """Ordena la lista con mezcla. Devuelve (lista, comparaciones, movimientos)."""
    contador = [0, 0]                                 # [comparaciones, movimientos]
    resultado = dividir_y_mezclar(lista, contador, mostrar, 0)
    return resultado, contador[0], contador[1]

# ---------- Ordenamiento Rápido (Quicksort) ----------

def particionar(lista, inicio, fin, contador):
    """Usa el elemento del medio como pivote; devuelve su posición final."""
    medio = (inicio + fin) // 2
    lista[medio], lista[fin] = lista[fin], lista[medio]   # el pivote pasa al final
    pivote = lista[fin]
    posicion = inicio
    for j in range(inicio, fin):
        contador[0] = contador[0] + 1
        if lista[j] < pivote:
            lista[posicion], lista[j] = lista[j], lista[posicion]
            contador[1] = contador[1] + 1
            posicion = posicion + 1
    lista[posicion], lista[fin] = lista[fin], lista[posicion]
    contador[1] = contador[1] + 1
    return posicion


def ordenar_parte(lista, inicio, fin, contador, mostrar):
    """Función recursiva: ordena la parte de la lista entre inicio y fin."""
    if inicio < fin:
        p = particionar(lista, inicio, fin, contador)
        if mostrar:
            print(f"  Pivote {lista[p]}: {lista[inicio:p]} < {lista[p]} < {lista[p + 1:fin + 1]}")
        ordenar_parte(lista, inicio, p - 1, contador, mostrar)
        ordenar_parte(lista, p + 1, fin, contador, mostrar)


def rapido(lista, mostrar):
    """Ordena la lista con quicksort. Devuelve (lista, comparaciones, movimientos)."""
    contador = [0, 0]                                     # [comparaciones, movimientos]
    ordenar_parte(lista, 0, len(lista) - 1, contador, mostrar)
    return lista, contador[0], contador[1]


def ordenar_con_sorted(lista, mostrar):
    """Referencia: sorted() usa Timsort, un algoritmo optimizado (mezcla + inserción)."""
    return sorted(lista), 0, 0

              ("Mezcla", mezcla), ("Rápido", rapido), ("sorted()", ordenar_con_sorted))
