# ============================================================
# Archivo: mision_orbita.py
# Curso: SOFT-01 Principios de Programación 1 - Sección SCV6
# Integrantes: Solución de referencia (docente)
# Fecha: 21/10/2026        Versión: 1.0
# Descripción: Sistema integrado Misión Órbita (versión individual): planifica,
#              ejecuta y reporta una misión completa.
# ============================================================

# ---------- Constantes ----------
DESTINOS = (("Luna", 3), ("Marte", 5), ("Europa", 7))
# (nombre, combustible, energía, provisiones, puntos)
ACTIVIDADES = (("Analizar terreno", 0, 8, 0, 10),
               ("Recolectar muestra", 0, 12, 2, 20),
               ("Realizar observación", 0, 5, 0, 8),
               ("Sobrevolar la zona", 15, 3, 0, 25))
RECURSOS = ("combustible", "oxígeno", "energía", "provisiones")
CONSUMO_COMBUSTIBLE_DIA = 8
CONSUMO_OXIGENO_DIA = 2          # por tripulante
CONSUMO_ENERGIA_DIA = 5
CONSUMO_PROVISIONES_DIA = 1      # por tripulante
TRAYECTOS = 2
RESERVA_COMBUSTIBLE = 10
OXIGENO_EMERGENCIA = 5
ENERGIA_EXPLORACION = 10
PROVISIONES_ADICIONALES = 3
MARGEN_MINIMO = 10
ACTIVIDAD_MUESTRA = "Recolectar muestra"


# ---------- A. Preparación y autorización ----------
def mostrar_destinos():
    """Muestra los destinos de la tupla DESTINOS."""
    print("Destinos disponibles:")
    numero = 1
    for destino in DESTINOS:
        nombre, dias = destino
        print(str(numero) + ". " + nombre + " (" + str(dias) + " días)")
        numero = numero + 1


def leer_recursos():
    """Lee los cuatro recursos disponibles y los devuelve en un diccionario."""
    recursos = {}
    for recurso in RECURSOS:
        recursos[recurso] = int(input("Cantidad disponible de " + recurso + ": "))
    return recursos


def calcular_requeridos(dias_viaje, tripulantes):
    """Devuelve un diccionario con los recursos requeridos (fórmulas del Avance 1)."""
    requeridos = {}
    requeridos["combustible"] = dias_viaje * CONSUMO_COMBUSTIBLE_DIA * TRAYECTOS + RESERVA_COMBUSTIBLE
    requeridos["oxígeno"] = dias_viaje * CONSUMO_OXIGENO_DIA * TRAYECTOS * tripulantes + OXIGENO_EMERGENCIA
    requeridos["energía"] = dias_viaje * CONSUMO_ENERGIA_DIA * TRAYECTOS + ENERGIA_EXPLORACION
    requeridos["provisiones"] = dias_viaje * tripulantes * CONSUMO_PROVISIONES_DIA * TRAYECTOS + PROVISIONES_ADICIONALES
    return requeridos


def mostrar_resumen(nombre_mision, nombre_destino, dias_viaje, tripulantes, disponibles, requeridos):
    """Muestra el resumen de planificación de la misión."""
    print()
    print("Misión:", nombre_mision)
    print("Destino:", nombre_destino)
    print("Duración del viaje de ida:", dias_viaje, "días")
    print("Tripulantes:", tripulantes)
    for recurso in RECURSOS:
        print(recurso.capitalize() + ": disponible " + str(disponibles[recurso])
              + " | requerido " + str(requeridos[recurso]))
    margen = disponibles["combustible"] - requeridos["combustible"]
    print("Margen adicional de combustible:", margen, "unidades")
    if margen < MARGEN_MINIMO:
        print("Advertencia: el margen adicional de combustible es bajo.")


def autorizar_lanzamiento(disponibles, requeridos):
    """Devuelve True si los cuatro recursos disponibles cubren lo requerido."""
    autorizado = True
    for recurso in RECURSOS:
        if disponibles[recurso] < requeridos[recurso]:
            autorizado = False
    return autorizado


# ---------- B. Viaje ----------
def calcular_consumo_diario(tripulantes):
    """Devuelve un diccionario con el consumo de un día de viaje."""
    consumo = {}
    consumo["combustible"] = CONSUMO_COMBUSTIBLE_DIA
    consumo["oxígeno"] = CONSUMO_OXIGENO_DIA * tripulantes
    consumo["energía"] = CONSUMO_ENERGIA_DIA
    consumo["provisiones"] = CONSUMO_PROVISIONES_DIA * tripulantes
    return consumo


def viajar_al_destino(recursos, dias_viaje, tripulantes):
    """Descuenta el consumo de todo el viaje de ida y devuelve los recursos actualizados."""
    consumo = calcular_consumo_diario(tripulantes)
    for recurso in RECURSOS:
        recursos[recurso] = recursos[recurso] - consumo[recurso] * dias_viaje
    return recursos


def recursos_alcanzan(recursos, consumo):
    """Devuelve True si cada recurso cubre su consumo diario."""
    alcanzan = True
    for recurso in RECURSOS:
        if recursos[recurso] < consumo[recurso]:
            alcanzan = False
    return alcanzan


def simular_trayecto(recursos, dias, tripulantes):
    """Simula el trayecto día a día y devuelve la cantidad de días completados."""
    consumo = calcular_consumo_diario(tripulantes)
    dias_completados = 0
    puede_continuar = True
    while dias_completados < dias and puede_continuar:
        # Se verifica antes de consumir: el día fallido no se descuenta.
        if recursos_alcanzan(recursos, consumo):
            for recurso in RECURSOS:
                recursos[recurso] = recursos[recurso] - consumo[recurso]
            dias_completados = dias_completados + 1
            linea = "Día " + str(dias_completados)
            for recurso in RECURSOS:
                linea = linea + " | " + recurso.capitalize() + ": " + str(recursos[recurso])
            print(linea)
        else:
